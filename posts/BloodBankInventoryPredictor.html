Blood bank inventory

1 Introduction

The app below provides an interactive tool for exploring a model of red blood cell inventory management research paper.

The models allows for the derivation of key performance indicators (KPIs).

The expressions for KPIs are as follows. The issuable stock index (ISI) is given by

\[ ISI=\frac{r}{k_2}-\frac{1}{k_1} \] The waste as a percentage of issuable stock (WAPI) is \[ WAPI=100e^{-k_2(A+\frac{1}{k_1}-\mu)}e^{r-1} \]

To estimate the KPIs, the following parameter values must first be estimated in a given clinical setting:

  • \(k_1\) - restock frequency
  • \(k_2\) - demand frequency
  • \(A\) - age that RBCs are disposed
  • \(\mu\) - mean age of supplied RBC
  • \(r\) - restock threshold (order up to value for a given blood group)

To aid parameter inference, in the app below the user is asked to define an observation time window and the number of demand events for a given blood group. These data are then used to estimate the demand frequency, \(k_2\).

def server(input, output, session):

    def ComputeIsi(supply_frequency,demand_freq,restock_threshold):

        rhs=restock_threshold/demand_freq-1.0/supply_frequency
        return rhs
    def ComputeWapi(supply_frequency,demand_freq,restock_threshold,mu,disposal_age):
        #100e^{-k_2(A+\frac{1}{k_1}-\mu)}e^{r-1}
        A=disposal_age
        k_2=demand_freq
        k_1=supply_frequency
        r=restock_threshold

        rhs=100.0*np.exp(-k_2*(A+1/k_1-mu))*np.exp(r-1)
        return rhs
    def ComputeMeanAgeTranUnit(supply_frequency,demand_freq,restock_threshold,mu,disposal_age):
        A=disposal_age
        k_2=demand_freq
        k_1=supply_frequency
        r=restock_threshold
        rhs=r/k_2-1.0/k_1+mu
        return rhs

    def ComputePredictedKPIs(supply_frequency,demand_freq,restock_threshold,mean_age_supplied,disposal_age):
        isi=ComputeIsi(supply_frequency,demand_freq,restock_threshold)
        wapi=ComputeWapi(supply_frequency,demand_freq,restock_threshold,mean_age_supplied,disposal_age)
        age_trans_init=ComputeMeanAgeTranUnit(supply_frequency,demand_freq,restock_threshold,mean_age_supplied,disposal_age)

        isi=np.round(isi)
        wapi=np.round(wapi,3)
        age_trans_init=np.round(age_trans_init,2)




        return isi,wapi,age_trans_init

    def ComputeOptimalReStockThresholds(supply_frequency,demand_freq,mean_age_supplied,disposal_age,target_ISI,target_WAPI,target_mean_age):  
        A=disposal_age
        k_2=demand_freq
        k_1=supply_frequency
        mu=mean_age_supplied


        r_max=k_2*(A+1.0/k_1-mu)

        r_wapi=r_max-1.0-np.log(100.0/target_WAPI)

        r_isi=k_2*(target_ISI+1.0/k_1)
        r_age_trans_init=r_max-k_2*(A-target_mean_age)

        r_wapi=np.round(r_wapi,1)
        r_isi=np.round(r_isi,1)
        r_age_trans_init=np.round(r_age_trans_init,1)




        return r_isi,r_wapi,r_age_trans_init

    @render.data_frame
    def displayKPIs():
        time_window=float(input.time_window())
        Num_used_units=float(input.Num_used_units())
        Num_supply_days=float(input.Num_supply_days())
        restock_threshold=float(input.Restock_threshold())

        mean_age_supplied=float(input.mean_age_supplied())
        disposal_age=float(input.disposal_age())

        



        supply_frequency=Num_supply_days/7.0
        demand_freq=Num_used_units/time_window

        isi,wapi,age_trans_init=ComputePredictedKPIs(supply_frequency,demand_freq,restock_threshold,mean_age_supplied,disposal_age)


        

        data_dict={'ISI': [isi],'Wapi': [wapi],'Mean_age':[age_trans_init]}
        df=pd.DataFrame.from_dict(data_dict)

        return render.DataTable(df,height=100)
        
    @render.data_frame
    def displayStockThresholds():
        time_window=float(input.time_window())
        Num_used_units=float(input.Num_used_units())
        Num_supply_days=float(input.Num_supply_days())
        mean_age_supplied=float(input.mean_age_supplied())
        disposal_age=float(input.disposal_age())

        target_ISI=float(input.target_ISI())
        target_WAPI=float(input.target_WAPI())
        target_mean_age=float(input.target_mean_age())

        supply_frequency=Num_supply_days/7.0
        demand_freq=Num_used_units/time_window

        r_isi,r_wapi,r_age_trans_init=ComputeOptimalReStockThresholds(supply_frequency,demand_freq,mean_age_supplied,disposal_age,target_ISI,target_WAPI,target_mean_age)

        

        data_dict={'Restock (ISI)': [r_isi],'Restock (WAPI)': [r_wapi],'Restock (Mean age)':[r_age_trans_init]}
        df=pd.DataFrame.from_dict(data_dict)

        return render.DataTable(df,height=100)  


    @render.data_frame
    def displayStockThresholdsAll():
        time_window=float(input.time_window())
        O_pos=float(input.O_pos())
        O_neg=float(input.O_neg())
        A_pos=float(input.A_pos())
        A_neg=float(input.A_neg())
        B_pos=float(input.B_pos())
        B_neg=float(input.B_neg())
        AB_pos=float(input.AB_pos())
        AB_neg=float(input.AB_neg())

        demand_vec={'O_pos':O_pos,'O_neg':O_neg,
        'A_pos':A_pos,
        'A_neg':A_neg,
        'B_pos':B_pos,
        'B_neg':B_neg,
        'AB_pos':AB_pos,
        'AB_neg':AB_neg}


        Num_supply_days=float(input.Num_supply_days())
        mean_age_supplied=float(input.mean_age_supplied())
        disposal_age=float(input.disposal_age())

        target_ISI=float(input.target_ISI_all())
        
        supply_frequency=Num_supply_days/7.0
        

        # Loop over dictionary and compute 
        #demand_freq=Num_used_units/time_window

        r_isi_all={}
        for key, value in demand_vec.items():
            demand_freq_i=value/time_window
            r_isi,r_wapi,r_age_trans_init=ComputeOptimalReStockThresholds(supply_frequency,demand_freq_i,mean_age_supplied,disposal_age,target_ISI,target_ISI,target_ISI)

            r_isi_all[key]=[int(r_isi)]


        

        #data_dict={'Restock (ISI)': [r_isi],'Restock (WAPI)': [r_wapi],'Restock (Mean age)':[r_age_trans_init]}
        df=pd.DataFrame.from_dict(r_isi_all)

        return render.DataTable(df)    
          
        

    


    
app = App(app_ui, server)
